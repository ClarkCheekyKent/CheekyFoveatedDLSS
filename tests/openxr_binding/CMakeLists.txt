cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CheekyOpenXRBindingRegression LANGUAGES CXX)
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
enable_testing()
if(NOT TARGET CheekyOpenXRLayer)
    add_library(CheekyOpenXRLayer SHARED "${root}/openxr_layer/openxr_layer.cpp" "${root}/shared/gaze_math.cpp"
        "${root}/openxr_layer/dlssg_worker.cpp" "${root}/openxr_layer/fg_focus.cpp")
    set(binding_layer_target CheekyOpenXRLayer)
endif()
add_executable(CheekyOpenXRBindingTests binding_tests.cpp "${root}/shared/gaze_math.cpp"
    "${root}/openxr_layer/dlssg_worker.cpp" "${root}/openxr_layer/fg_focus.cpp")
foreach(target ${binding_layer_target} CheekyOpenXRBindingTests)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_include_directories(${target} PRIVATE "${root}/shared" "${root}/third_party/openxr/include")
    target_link_libraries(${target} PRIVATE d3d11 d3d12 d3dcompiler dxgi dxguid version)
    target_link_libraries(${target} PRIVATE "${root}/third_party/ngx-sdk/nvsdk_ngx_s$<$<CONFIG:Debug>:_dbg>.lib")
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    endif()
endforeach()
set_target_properties(CheekyOpenXRBindingTests PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
add_test(NAME OpenXRHostBindings COMMAND CheekyOpenXRBindingTests)
add_test(NAME OpenXRFGCopyStates COMMAND CheekyOpenXRBindingTests --fg-gpu-validation)
add_test(NAME OpenXRFGCropRecenter COMMAND CheekyOpenXRBindingTests --fg-crop-gpu-validation)
add_test(NAME OpenXRFGHardening COMMAND CheekyOpenXRBindingTests --fg-hardening)
